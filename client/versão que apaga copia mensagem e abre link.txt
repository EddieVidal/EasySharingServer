# API Nativas do Windows para Notificação/Piscar Barra de Tarefas
FLASHW_ALL = 3
FLASHW_TIMERNOFG = 12

class FLASHWINFO(ctypes.Structure):
    _fields_ = [
        ('cbSize', ctypes.c_uint),
        ('hwnd', ctypes.c_void_p),
        ('dwFlags', ctypes.c_uint),
        ('uCount', ctypes.c_uint),
        ('dwTimeout', ctypes.c_uint)
    ]

def flash_window(hwnd):
    """Pisca o ícone do aplicativo na barra de tarefas do Windows."""
    try:
        finfo = FLASHWINFO()
        finfo.cbSize = ctypes.sizeof(FLASHWINFO)
        finfo.hwnd = hwnd
        finfo.dwFlags = FLASHW_ALL | FLASHW_TIMERNOFG
        finfo.uCount = 5
        finfo.dwTimeout = 0
        ctypes.windll.user32.FlashWindowEx(ctypes.byref(finfo))
    except Exception:
        pass

def normalize_id(user_id: str) -> str:
    clean = user_id.strip().upper()
    if not clean.startswith("USER-"):
        clean = f"USER-{clean}"
    return clean


def get_hardware_uuid() -> str:
    try:
        key = winreg.OpenKey(
            winreg.HKEY_LOCAL_MACHINE,
            r"SOFTWARE\Microsoft\Cryptography",
            0,
            winreg.KEY_READ | winreg.KEY_WOW64_64KEY
        )
        value, _ = winreg.QueryValueEx(key, "MachineGuid")
        winreg.CloseKey(key)
        
        if value:
            short_hash = hashlib.sha256(value.encode()).hexdigest()[:8].upper()
            return f"USER-{short_hash}"
    except Exception:
        pass
    return "USER-" + secrets.token_hex(4).upper()

# Regex simples para detectar URLs http(s) dentro de uma mensagem de texto
URL_REGEX = re.compile(r'(https?://[^\s]+)')


def _messages_file_path(target_key: str) -> Path:
    safe = target_key.replace(":", "_").replace("/", "_")
    return MESSAGES_DIR / f"{safe}.dat"


def load_chat_history_from_disk(target_key: str, key: bytes) -> list:
    """Carrega e decifra o histórico salvo localmente para uma conversa."""
    path = _messages_file_path(target_key)
    if not path.exists():
        return []
    try:
        raw_payload = json.loads(path.read_text(encoding="utf-8"))
        decrypted = crypto_utils.decrypt_text(raw_payload, key)
        history = json.loads(decrypted)
        return history if isinstance(history, list) else []
    except Exception as exc:
        print(f"Erro ao carregar histórico local de {target_key}: {exc}")
        return []


def save_chat_history_to_disk(target_key: str, key: bytes, history: list) -> None:
    """Cifra e salva o histórico de uma conversa em disco (persiste entre sessões)."""
    try:
        payload = crypto_utils.encrypt_text(json.dumps(history, ensure_ascii=False), key)
        _messages_file_path(target_key).write_text(json.dumps(payload), encoding="utf-8")
    except Exception as exc:
        print(f"Erro ao salvar histórico local de {target_key}: {exc}")


def delete_chat_history_file(target_key: str) -> None:
    """Remove o arquivo de histórico local de uma conversa do disco."""
    try:
        _messages_file_path(target_key).unlink(missing_ok=True)
    except Exception:
        pass


def format_size(num_bytes: int) -> str:
    size = float(num_bytes)
    for unit in ["B", "KB", "MB", "GB"]:
        if size < 1024:
            return f"{size:.1f} {unit}"
        size /= 1024
    return f"{size:.1f} TB"


def derive_channel_and_key(my_id: str, contact_id: str) -> tuple[str, bytes]:
    id1 = normalize_id(my_id)
    id2 = normalize_id(contact_id)
    ids = sorted([id1, id2])
    combined = f"{ids[0]}:{ids[1]}"
    channel_id = hashlib.sha256(combined.encode()).hexdigest()[:16]
    key = hashlib.sha256((combined + "_secret_salt").encode()).digest()
    return channel_id, key

def make_square_avatar(img: Image.Image, size: int = AVATAR_SIZE) -> Image.Image:
    """Recorta a imagem no centro (quadrado) e redimensiona para `size`x`size`."""
    img = img.convert("RGB")
    w, h = img.size
    m = min(w, h)
    left = (w - m) // 2
    top = (h - m) // 2
    img = img.crop((left, top, left + m, top + m))
    return img.resize((size, size), Image.LANCZOS)


def create_status_icon(status: str, size: int = 18) -> Image.Image:
    """Cria um ícone de presença em PNG/RGBA usando Pillow."""
    img = Image.new("RGBA", (size, size), (0, 0, 0, 0))

    margin = max(1, size // 9)
    box = (margin, margin, size - margin - 1, size - margin - 1)

    if status == "online":
        draw = ImageDraw.Draw(img)
        draw.ellipse(box, fill="#2fb344")

    elif status == "away":
        outer = Image.new("L", (size, size), 0)
        inner = Image.new("L", (size, size), 0)
        outer_draw = ImageDraw.Draw(outer)
        inner_draw = ImageDraw.Draw(inner)

        outer_draw.ellipse(box, fill=255)

        shift = max(3, size // 3)
        inner_box = (
            margin + shift,
            margin - shift // 3,
            size - margin + shift,
            size - margin - shift // 3,
        )
        inner_draw.ellipse(inner_box, fill=255)

        crescent = ImageChops.subtract(outer, inner)
        color = Image.new("RGBA", (size, size), "#d9d9d9")
        img = Image.composite(color, img, crescent)

    elif status == "busy":
        draw = ImageDraw.Draw(img)
        draw.ellipse(box, fill="#d9453d")

        bar_height = max(2, size // 5)
        y1 = (size - bar_height) // 2
        y2 = y1 + bar_height - 1
        draw.rectangle(
            (margin + 3, y1, size - margin - 3, y2),
            fill="#ffffff"
        )

    elif status == "offline":
        draw = ImageDraw.Draw(img)
        draw.ellipse(box, fill="#808080")

    return img

class CTkToolTip:
    """Implementação ultra-estável de Dica Flutuante (Tooltip)."""
    def __init__(self, widget, text):
        self.widget = widget
        self.text = text
        self.tooltip_window = None
        self._id = None

        self.widget.bind("<Enter>", self.schedule_show, add="+")
        self.widget.bind("<Leave>", self.hide_tooltip, add="+")
        self.widget.bind("<ButtonPress>", self.hide_tooltip, add="+")

    def schedule_show(self, event=None):
        self.unschedule()
        self._id = self.widget.after(200, self.show_tooltip)

    def unschedule(self):
        if self._id:
            self.widget.after_cancel(self._id)
            self._id = None

    def show_tooltip(self, event=None):
        if self.tooltip_window or not self.text:
            return

        x = self.widget.winfo_rootx() + 20
        y = self.widget.winfo_rooty() + 30

        self.tooltip_window = tw = ctk.CTkToplevel(self.widget)
        tw.wm_overrideredirect(True)
        tw.wm_geometry(f"+{x}+{y}")
        tw.attributes("-topmost", True)

        label = ctk.CTkLabel(
            tw,
            text=self.text,
            font=ctk.CTkFont(size=11),
            fg_color="#1f1f1f",
            text_color="#ffffff",
            corner_radius=4,
            padx=6,
            pady=3
        )
        label.pack()

    def hide_tooltip(self, event=None):
        self.unschedule()
        if self.tooltip_window:
            try:
                self.tooltip_window.destroy()
            except Exception:
                pass
            self.tooltip_window = None

    def _load_all_histories(self):
        """Carrega do disco o histórico de cada contato/grupo já cadastrado."""
        for name, contact_id in self.contacts.items():
            target_key = f"contact:{name}"
            _, key = derive_channel_and_key(self.my_id, contact_id)
            self.chat_history[target_key] = load_chat_history_from_disk(target_key, key)

        for gid, ginfo in self.groups.items():
            target_key = f"group:{gid}"
            try:
                key = base64.b64decode(ginfo["key"])
            except Exception:
                continue
            self.chat_history[target_key] = load_chat_history_from_disk(target_key, key)

    def _get_channel_key(self, target_key: str):
        """Retorna a chave de cifra de um canal, mesmo sem conexão ativa no momento."""
        conn = self.connections.get(target_key)
        if conn and conn.get("key"):
            return conn["key"]

        if ":" not in target_key:
            return None
        chat_type, target = target_key.split(":", 1)
        if chat_type == "contact" and target in self.contacts:
            _, key = derive_channel_and_key(self.my_id, self.contacts[target])
            return key
        if chat_type == "group" and target in self.groups:
            try:
                return base64.b64decode(self.groups[target]["key"])
            except Exception:
                return None
        return None

    def _persist_history(self, target_key: str):
        """Salva em disco (cifrado) o histórico atual de uma conversa."""
        key = self._get_channel_key(target_key)
        if key is None:
            return
        save_chat_history_to_disk(target_key, key, self.chat_history.get(target_key, []))

    def _wake_up_server(self):
        def worker():
            server_url = self.config_data["server_url"].rstrip("/")
            try:
                requests.get(f"{server_url}/health", timeout=15)
            except Exception:
                pass

        threading.Thread(target=worker, daemon=True).start()

    def _pick_profile_photo(self):
        path = filedialog.askopenfilename(
            title="Escolher foto de perfil",
            filetypes=[("Imagens", "*.png *.jpg *.jpeg *.webp *.bmp")]
        )
        if not path:
            return
        try:
            img = Image.open(path)
            square = make_square_avatar(img, AVATAR_SIZE)
            square.save(MY_AVATAR_PATH, format="JPEG", quality=82)
            self.my_avatar_bytes = MY_AVATAR_PATH.read_bytes()
            self.my_avatar_image = ctk.CTkImage(light_image=square, dark_image=square, size=(MY_AVATAR_THUMB, MY_AVATAR_THUMB))
            self.my_avatar_btn.configure(image=self.my_avatar_image, text="")
            self._broadcast_my_avatar()
            messagebox.showinfo("Foto atualizada", "Sua foto de perfil foi atualizada e enviada aos contatos conectados agora.")
        except Exception as exc:
            messagebox.showerror("Erro", f"Não foi possível carregar a imagem: {exc}")

    def _broadcast_my_avatar(self):
        if not self.my_avatar_bytes:
            return
        b64 = base64.b64encode(self.my_avatar_bytes).decode("ascii")
        for conn in list(self.connections.values()):
            if conn.get("type") == "contact" and conn.get("connected") and conn.get("ws"):
                try:
                    payload = crypto_utils.encrypt_text(b64, conn["key"])
                    conn["ws"].send(json.dumps({"type": "profile_photo", "payload": payload, "sender": self.my_id}))
                except Exception:
                    pass

    def _edit_contact(self, old_name: str):
        old_id = self.contacts.get(old_name, "")
        dialog = ContactDialog(self, title="Editar Contato", initial_name=old_name, initial_id=old_id)
        self.wait_window(dialog)
        if dialog.result:
            new_name, new_id = dialog.result
            
            old_key = f"contact:{old_name}"
            self._close_connection(old_key)
            del self.contacts[old_name]

            new_key = f"contact:{new_name}"
            if old_key in self.chat_history:
                self.chat_history[new_key] = self.chat_history.pop(old_key)
            if old_key in self.unread_counts:
                self.unread_counts[new_key] = self.unread_counts.pop(old_key)
            if old_key in self.contact_status:
                self.contact_status[new_key] = self.contact_status.pop(old_key)
            if old_key in self.contact_avatar_photos:
                self.contact_avatar_photos[new_key] = self.contact_avatar_photos.pop(old_key)
            old_avatar_path = self._avatar_cache_path(old_key)
            if old_avatar_path.exists():
                try:
                    old_avatar_path.replace(self._avatar_cache_path(new_key))
                except OSError:
                    pass

            self.contacts[new_name] = new_id
            save_contacts(self.contacts)

            if self.active_chat_type == "contact" and self.active_chat_target == old_name:
                self.active_chat_target = new_name

            self._start_single_connection("contact", new_name)
            self._render_sidebar_list()

    def _delete_contact(self, name: str):
        if messagebox.askyesno("Excluir Contato", f"Tem certeza que deseja remover {name}?", parent=self):
            target_key = f"contact:{name}"
            self._close_connection(target_key)

            if name in self.contacts:
                del self.contacts[name]
                save_contacts(self.contacts)

            self.chat_history.pop(target_key, None)
            self.unread_counts.pop(target_key, None)
            self.contact_status.pop(target_key, None)
            self.contact_avatar_photos.pop(target_key, None)
            try:
                self._avatar_cache_path(target_key).unlink(missing_ok=True)
            except Exception:
                pass

            if self.active_chat_type == "contact" and self.active_chat_target == name:
                self.active_chat_type = None
                self.active_chat_target = None
                self.chat_header.configure(text="Selecione uma conversa.")
                self.msg_entry.configure(state="disabled")
                self.send_btn.configure(state="disabled")
                self.attach_btn.configure(state="disabled")
                self.delete_chat_btn.configure(state="disabled")
                for widget in self.msg_scroll.winfo_children():
                    widget.destroy()

            self._render_sidebar_list()

    def _create_group(self):
        if not self.contacts:
            messagebox.showinfo("Aviso", "Você precisa adicionar contatos primeiro para criar um grupo.")
            return

        dialog = CreateGroupDialog(self, self.contacts)
        self.wait_window(dialog)

        if dialog.result:
            gname, selected_ids = dialog.result
            members = [self.my_id] + selected_ids
            server_url = self.config_data["server_url"].rstrip("/")

            def worker():
                try:
                    res = requests.post(f"{server_url}/groups", json={"group_name": gname, "members": members}, timeout=10)
                    res.raise_for_status()
                    gdata = res.json()
                    self.groups[gdata["group_id"]] = gdata
                    save_groups(self.groups)
                    self.after(0, lambda: self._start_single_connection("group", gdata["group_id"]))
                    self.after(0, self._render_sidebar_list)
                    self.after(0, lambda: messagebox.showinfo("Sucesso", f"Grupo '{gname}' criado com sucesso!"))
                except Exception as exc:
                    self.after(0, lambda: messagebox.showerror("Erro", f"Falha ao criar grupo no servidor: {exc}"))

            threading.Thread(target=worker, daemon=True).start()

    def _sync_groups_from_server(self):
        def worker():
            try:
                server_url = self.config_data["server_url"].rstrip("/")
                res = requests.get(f"{server_url}/groups/{self.my_id}", timeout=10)
                if res.status_code == 200:
                    remote_groups = res.json()
                    for g in remote_groups:
                        gid = g["group_id"]
                        if gid not in self.groups:
                            self.groups[gid] = g
                            self.after(0, lambda gid=gid: self._start_single_connection("group", gid))
                    save_groups(self.groups)
                    self.after(0, self._render_sidebar_list)
            except Exception:
                pass

        threading.Thread(target=worker, daemon=True).start()

    def _initialize_all_connections(self):

    def _start_single_connection(self, chat_type: str, target: str):
        target_key = f"{chat_type}:{target}"
        existing = self.connections.get(target_key)
        if existing and existing.get("state") in ("connecting", "connected"):
            return

        if chat_type == "contact":
            if target not in self.contacts:
                return
            contact_id = self.contacts[target]
            channel_id, key = derive_channel_and_key(self.my_id, contact_id)
        else:
            if target not in self.groups:
                return
            ginfo = self.groups[target]
            channel_id = target
            key = base64.b64decode(ginfo["key"])

        generation = (existing["generation"] + 1) if existing else 1
        conn_entry = {
            "ws": None,
            "key": key,
            "channel_id": channel_id,
            "connected": False,
            "state": "connecting",
            "type": chat_type,
            "target": target,
            "generation": generation,
        }
        self.connections[target_key] = conn_entry

        server_url = self.config_data["server_url"].rstrip("/")
        ws_url = server_url.replace("https://", "wss://").replace("http://", "ws://") + f"/ws/{channel_id}"

        def is_current() -> bool:
            live = self.connections.get(target_key)
            return bool(live) and live.get("generation") == generation

        def on_message(ws, message):
            if not is_current():
                return
            try:
                data = json.loads(message)
                sender = data.get("sender", "")
                
                if sender == self.my_id:
                    return

                if data.get("type") == "text":
                    decrypted_text = crypto_utils.decrypt_text(data["payload"], key)
                    msg_display = f"{sender}: {decrypted_text}" if (chat_type == "group" and sender) else decrypted_text
                    self.after(0, lambda: self._handle_incoming_message(target_key, "text", text=msg_display, is_me=False))
                
                elif data.get("type") == "file":
                    code = data["code"]
                    filename = data["filename"]
                    size = data["size"]
                    self.after(0, lambda: self._handle_incoming_message(target_key, "file", code=code, filename=filename, size=size, is_me=False, sender=sender))

                elif data.get("type") == "presence" and chat_type == "contact":
                    status = data.get("status", "offline")
                    self.after(0, lambda: self._on_presence_received(target_key, status))

                elif data.get("type") == "profile_photo" and chat_type == "contact":
                    b64 = crypto_utils.decrypt_text(data["payload"], key)
                    raw = base64.b64decode(b64)
                    self.after(0, lambda: self._on_avatar_received(target_key, raw))

                elif data.get("type") == "profile_photo_removed" and chat_type == "contact":
                    self.after(0, lambda: self._on_avatar_removed(target_key))

                elif data.get("type") == "sync_request" and chat_type == "contact":
                    try:
                        ws.send(json.dumps({"type": "presence", "status": self.my_status, "sender": self.my_id}))
                        if self.my_avatar_bytes:
                            avatar_b64 = base64.b64encode(self.my_avatar_bytes).decode("ascii")
                            avatar_payload = crypto_utils.encrypt_text(avatar_b64, key)
                            ws.send(json.dumps({"type": "profile_photo", "payload": avatar_payload, "sender": self.my_id}))
                        else:
                            ws.send(json.dumps({"type": "profile_photo_removed", "sender": self.my_id}))
                    except Exception:
                        pass
            except Exception as exc:
                print(f"Erro ao processar mensagem em {target_key}: {exc}")

        def on_open(ws):
            if not is_current():
                return
            conn_entry["connected"] = True
            conn_entry["state"] = "connected"
            try:
                ws.send(json.dumps({"type": "hello", "user_id": self.my_id}))
                if chat_type == "contact":
                    ws.send(json.dumps({"type": "presence", "status": self.my_status, "sender": self.my_id}))
                    if self.my_avatar_bytes:
                        b64 = base64.b64encode(self.my_avatar_bytes).decode("ascii")
                        payload = crypto_utils.encrypt_text(b64, key)
                        ws.send(json.dumps({"type": "profile_photo", "payload": payload, "sender": self.my_id}))
                    else:
                        ws.send(json.dumps({"type": "profile_photo_removed", "sender": self.my_id}))
                    ws.send(json.dumps({"type": "sync_request", "sender": self.my_id}))
            except Exception as exc:
                print(f"Erro no handshake (hello/presença/avatar): {exc}")
            if self.active_chat_type == chat_type and self.active_chat_target == target:
                self.after(0, lambda: self.msg_entry.configure(state="normal"))
                self.after(0, lambda: self.send_btn.configure(state="normal"))
                self.after(0, lambda: self.attach_btn.configure(state="normal"))
            self.after(0, self._refresh_header_status)

        def on_error(ws, error):
            if not is_current():
                return
            conn_entry["connected"] = False
            conn_entry["state"] = "error"
            self._last_error_detail = str(error)
            self.after(0, self._refresh_header_status)

        def on_close(ws, status_code, msg):
            if not is_current():
                return
            conn_entry["connected"] = False
            conn_entry["state"] = "disconnected"
            self.after(0, self._refresh_header_status)
            if chat_type == "contact":
                self.after(0, lambda: self._mark_contact_possibly_offline(target_key))
            self.after(5000, lambda: self._start_single_connection(chat_type, target))

        def run():
            ws = websocket.WebSocketApp(ws_url, on_message=on_message, on_open=on_open, on_error=on_error, on_close=on_close)
            conn_entry["ws"] = ws
            ws.run_forever(ping_interval=30, ping_timeout=10)

        thread = threading.Thread(target=run, daemon=True)
        conn_entry["thread"] = thread
        thread.start()

    def _close_connection(self, target_key: str):
        if target_key in self.connections:
            conn = self.connections.pop(target_key)
            conn["generation"] = conn.get("generation", 0) + 1
            if conn.get("ws"):
                try:
                    conn["ws"].close()
                except Exception:
                    pass

    def _handle_incoming_message(self, target_key: str, msg_type: str, **kwargs):
        if target_key not in self.chat_history:
            self.chat_history[target_key] = []
        
        hist_item = {"id": uuid.uuid4().hex, "ts": time.time(), "type": msg_type, **kwargs}
        self.chat_history[target_key].append(hist_item)
        self._persist_history(target_key)

        is_current_active = (
            self.active_chat_type and self.active_chat_target and 
            f"{self.active_chat_type}:{self.active_chat_target}" == target_key
        )

        if not is_current_active:
            self.unread_counts[target_key] = self.unread_counts.get(target_key, 0) + 1
            
            try:
                winsound.PlaySound("SystemNotification", winsound.SND_ALIAS | winsound.SND_ASYNC)
            except Exception:
                pass

            try:
                hwnd = ctypes.windll.user32.GetParent(self.winfo_id())
                if not hwnd:
                    hwnd = self.winfo_id()
                flash_window(hwnd)
            except Exception:
                pass

            self._update_app_title()
            self._render_sidebar_list()
        else:
            if msg_type == "text":
                self._add_chat_bubble(kwargs["text"], kwargs["is_me"], msg_id=hist_item["id"], target_key=target_key, render_only=True)
            elif msg_type == "file":
                self._add_file_bubble(kwargs["code"], kwargs["filename"], kwargs["size"], kwargs["is_me"], kwargs.get("sender", ""), msg_id=hist_item["id"], target_key=target_key, render_only=True)

    def _update_app_title(self):
        total_unread = sum(self.unread_counts.values())
        base_title = f"Easy Sharing — Chat V.{APP_VERSION} - Por Edson Vidal."
        if total_unread > 0:
            self.title(f"🔴 ({total_unread}) {base_title}")
        else:
            self.title(base_title)

    def _refresh_header_status(self):
        if self.active_chat_type and self.active_chat_target:
            target_key = f"{self.active_chat_type}:{self.active_chat_target}"
            conn = self.connections.get(target_key)
            if conn and conn.get("connected"):
                self._set_connection_status("connected")
            elif conn and conn.get("state") == "error":
                self._set_connection_status("error", getattr(self, "_last_error_detail", ""))
            else:
                self._set_connection_status("connecting")
            return

        total = len(self.connections)
        connected = sum(1 for c in self.connections.values() if c.get("connected"))
        if total == 0:
            self.status_dot.configure(text_color="gray")
            self.status_text.configure(text="")
        elif connected > 0:
            self.status_dot.configure(text_color="#2fb344")
            self.status_text.configure(text=f"Servidor online ({connected}/{total})")
        else:
            self.status_dot.configure(text_color="#e2b93b")
            self.status_text.configure(text="Conectando ao servidor...")

    def _set_connection_status(self, status: str, detail: str = ""):
        colors = {
            "connecting": ("#e2b93b", "Conectando..."),
            "connected": ("#2fb344", "Conectado"),
            "disconnected": ("#8a8a8a", "Desconectado"),
            "error": ("#d9453d", "Erro de conexão"),
        }
        color, label = colors.get(status, ("gray", ""))
        self.status_dot.configure(text_color=color)
        self.status_text.configure(text=f"{label}{(' — ' + detail) if detail else ''}")

    def _send_message(self):
        text = self.msg_entry.get().strip()
        if not text or not self.active_chat_target or not self.active_chat_type:
            return

        target_key = f"{self.active_chat_type}:{self.active_chat_target}"
        conn_data = self.connections.get(target_key)

        if not conn_data or not conn_data.get("ws") or not conn_data.get("connected"):
            messagebox.showwarning("Sem Conexão", "Aguarde a conexão deste chat ser estabelecida.")
            return

        self.msg_entry.delete(0, "end")
        payload = crypto_utils.encrypt_text(text, conn_data["key"])
        msg_data = json.dumps({"type": "text", "payload": payload, "sender": self.my_id})

        try:
            conn_data["ws"].send(msg_data)
            if target_key not in self.chat_history:
                self.chat_history[target_key] = []
            msg_id = uuid.uuid4().hex
            self.chat_history[target_key].append({"id": msg_id, "ts": time.time(), "type": "text", "text": text, "is_me": True})
            self._persist_history(target_key)
            self._add_chat_bubble(text, is_me=True, msg_id=msg_id, target_key=target_key, render_only=True)
        except Exception as exc:
            self._set_connection_status("error", str(exc))

    @staticmethod
    def _extract_first_url(text: str) -> str | None:
        match = URL_REGEX.search(text)
        if not match:
            return None
        # Remove pontuação de fechamento que costuma grudar no fim do link (. , ; ) ! ?)
        return match.group(1).rstrip(').,;!?"\'')

            self._rerender_active_chat()

    def _attach_and_send_file(self):
        path = filedialog.askopenfilename(title="Selecione um arquivo para enviar")
        if not path or not self.active_chat_target or not self.active_chat_type:
            return

        target_key = f"{self.active_chat_type}:{self.active_chat_target}"
        conn_data = self.connections.get(target_key)

        if not conn_data or not conn_data.get("ws") or not conn_data.get("connected"):
            messagebox.showwarning("Sem Conexão", "Conexão inativa para este chat.")
            return

        def worker():
            filename = os.path.basename(path)
            server_url = self.config_data["server_url"].rstrip("/")
            size = os.path.getsize(path)

            key = crypto_utils.generate_key()
            tmp_encrypted = tempfile.NamedTemporaryFile(delete=False, suffix=".enc")
            tmp_encrypted.close()
            enc_path = tmp_encrypted.name

            try:
                crypto_utils.encrypt_file(path, enc_path, key, filename)
                with open(enc_path, "rb") as f:
                    encoder = MultipartEncoder(fields={"file": (GENERIC_UPLOAD_NAME, f, "application/octet-stream")})
                    response = requests.post(f"{server_url}/upload", data=encoder, headers={"Content-Type": encoder.content_type}, timeout=(20, 120))
                response.raise_for_status()
                code = response.json()["code"]

                key_text = base64.urlsafe_b64encode(key).decode().rstrip("=")
                full_code = f"{code}.{key_text}"
                file_msg = json.dumps({"type": "file", "code": full_code, "filename": filename, "size": size, "sender": self.my_id})
                
                conn_data["ws"].send(file_msg)

                if target_key not in self.chat_history:
                    self.chat_history[target_key] = []
                file_msg_id = uuid.uuid4().hex
                self.chat_history[target_key].append({"id": file_msg_id, "ts": time.time(), "type": "file", "code": full_code, "filename": filename, "size": size, "is_me": True})
                self._persist_history(target_key)

                self.after(0, lambda: self._add_file_bubble(full_code, filename, size, True, msg_id=file_msg_id, target_key=target_key, render_only=True))
            except Exception as exc:
                self.after(0, lambda: messagebox.showerror("Erro", f"Falha ao enviar arquivo: {exc}"))
            finally:
                try:
                    os.remove(enc_path)
                except OSError:
                    pass

        threading.Thread(target=worker, daemon=True).start()

    def _download_attached_file(self, full_code: str, original_filename: str):
        save_path = filedialog.asksaveasfilename(initialfile=original_filename)
        if not save_path:
            return

        def worker():
            server_url = self.config_data["server_url"].rstrip("/")
            if "." in full_code:
                code, key_text = full_code.split(".", 1)
                padding = "=" * (-len(key_text) % 4)
                key = base64.urlsafe_b64decode(key_text + padding)
            else:
                code = full_code
                target_key = f"{self.active_chat_type}:{self.active_chat_target}"
                key = self.connections[target_key]["key"]

            tmp_encrypted = tempfile.NamedTemporaryFile(delete=False, suffix=".enc")
            tmp_encrypted.close()
            enc_path = tmp_encrypted.name
            tmp_decrypted_path = enc_path + ".dec"

            try:
                with requests.get(f"{server_url}/download/{code}", stream=True, timeout=(20, 60)) as response:
                    response.raise_for_status()
                    with open(enc_path, "wb") as f:
                        for chunk in response.iter_content(chunk_size=1024 * 128):
                            if chunk:
                                f.write(chunk)

                crypto_utils.decrypt_file(enc_path, tmp_decrypted_path, key)
                os.replace(tmp_decrypted_path, save_path)
                self.after(0, lambda: messagebox.showinfo("Sucesso", f"Arquivo salvo em:\n{save_path}"))
            except Exception as exc:
                self.after(0, lambda: messagebox.showerror("Erro", f"Falha no download: {exc}"))
            finally:
                try:
                    os.remove(enc_path)
                except OSError:
                    pass

        threading.Thread(target=worker, daemon=True).start()
